ClrHome
Menu("SOLVE","QUAD",1,"SIMPLIFY",2,"RATIONALIZE",3)

Lbl 1
Input "A:",A
Input "B:",B
Input "C:",C
(B²-4*A*C)→N
If N<0
Then
Disp "NON-REAL"
Stop
End
((0-B)+√(N))/(2*A)→D
((0-B)-√(N))/(2*A)→E
Disp "X=",D,E
Stop

Lbl 2
Input "VALUE:",N
1→H
round(N²,0)→I
For(S,int(abs(N)+.1),1,0-1)
If fPart(I/S²)=0
Then
S→H
I/S²→I
1→S
End
End
If I=1
Then
Disp "SIMP: "+toString(H)
Else
If H=1
Then
Disp "SIMP: √("+toString(I)+")"
Else
Disp "SIMP: "+toString(H)+"√("+toString(I)+")"
End
End
Stop

Lbl 3
Input "NUMERATOR:",A
Input "RAD DENOM:",N
1→H
round(N²,0)→I
For(S,int(abs(N)+.1),1,0-1)
If fPart(I/S²)=0
Then
S→H
I/S²→I
1→S
End
End
A→U
(H*I)→V
gcd(U,V)→G
U/G→U
V/G→V
Disp "RESULT:"
If V=1
Then
Disp toString(U)+"√("+toString(I)+")"
Else
Disp toString(U)+"√("+toString(I)+")"
Disp "-------"
Disp toString(V)
End
Stop
