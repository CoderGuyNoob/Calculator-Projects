ClrHome
Menu("SOLVE","QUAD",1,"SIMPLIFY",2,"RATIONALIZE",3)

Lbl 1
Input "A:",A
Input "B:",B
Input "C:",C
(B²-4*A*C)→N
If N<0
Then
Disp "NON-REAL"
Stop
End
((0-B)+√(N))/(2*A)→D
((0-B)-√(N))/(2*A)→E
Disp "X=",D,E
Stop


Lbl 2
Input "VALUE:",N
0→T
Goto 9


Lbl 3
Input "CONST NUM:",A
Input "RAD NUM:",B
Input "CONST DEN:",C
Input "RAD DEN:",D
Input "RADICAND:",N

1→T
Goto 9


Lbl 3.5
A*C-B*D*N→U
B*C-A*D→V
C²-D²*N→W

Disp "RESULT:"
Disp toString(U)+"+"+toString(V)+"√("+toString(I)+")"
Disp "------"
Disp toString(W)
Stop


Lbl 9
1→H
round(N²,0)→I
For(S,int(abs(N)+.1),1,0-1)
If fPart(I/S²)=0
Then
S→H
I/S²→I
1→S
End
End

If T=1
Then
Goto 3.5
End

If I=1
Then
Disp "SIMP: "+toString(H)
Else
If H=1
Then
Disp "SIMP: √("+toString(I)+")"
Else
Disp "SIMP: "+toString(H)+"√("+toString(I)+")"
End
End
Stop
