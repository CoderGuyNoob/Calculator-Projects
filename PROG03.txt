ClrHome
Menu("SOLVE","QUAD",1,"SIMPLIFY √",2)

Lbl 1
Input "A:",A
Input "B:",B
Input "C:",C
(B²-4*A*C)→N
If N<0
Then
Disp "NON-REAL"
Stop
End
((0-B)+√(N))/(2*A)→D
((0-B)-√(N))/(2*A)→E
Disp "X=",D,E
Stop

Lbl 2
Input "VALUE:",N
1→H
N→I
For(S,int(√(abs(N))+.1),1,0-1)
If fPart(N/S²)=0
Then
S→H
N/S²→I
1→S
End
End
If I=1
Then
Disp "SIMP: "+toString(H)
Else
If H=1
Then
Disp "SIMP: √("+toString(I)+")"
Else
Disp "SIMP: "+toString(H)+"√("+toString(I)+")"
End
End
