Menu("COORD TOOLS","DIST",1,"MID",2,"END",3,"SLOPE",4)

Lbl 1
Input "X1=",A
Input "Y1=",B
Input "X2=",C
Input "Y2=",D
(C-A)²+(D-B)²→N
Goto SIMPLIFY       // Simplify distance
Stop

Lbl 2
Input "X1=",A
Input "Y1=",B
Input "X2=",C
Input "Y2=",D
(A+C)/2→E
(B+D)/2→F
Disp "MIDPOINT"
Disp "("+toString(E)+","+toString(F)+")"
Stop

Lbl 3
Input "X1=",A
Input "Y1=",B
Input "X2=",C
Input "Y2=",D
2C-A→E
2D-B→F
Disp "ENDPOINT"
Disp "("+toString(E)+","+toString(F)+")"
Stop

Lbl 4
Input "X1=",A
Input "Y1=",B
Input "X2=",C
Input "Y2=",D
(D-B)/(C-A)→E
Disp "SLOPE="+toString(E)
Stop

Lbl SIMPLIFY
1→H
N→I
For(S,1,int(√(N)+.1))
If 0=fPart(round(N/S²,8))
Then
S→H
N/S²→I
End
End
If I=1
Then
Disp toString(H)
Else
If H=1
Then
Disp "√("+toString(I)+")"
Else
Disp toString(H)+"√("+toString(I)+")"
End
End
Stop
