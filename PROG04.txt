ClrHome
Menu("TRI","TYPE",1,"SIM",2)

Lbl 1
Input "A=",A
Input "B=",B
Input "C=",C

{A,B,C}→L₁
SortA(L₁)

L₁(1)→A
L₁(2)→B
L₁(3)→C

If A+B≤C
Then
Disp "NOT TRI"
Stop
End

A²+B²→D

If D=C²
Then
Disp "R"
Else
If D<C²
Then
Disp "O"
Else
Disp "A"
End
End

If A=C
Then
Disp "E"
Else
If A=B or B=C
Then
Disp "I"
Else
Disp "S"
End
End
Stop

Lbl 2
Input "A=",A
Input "B=",B
Input "C=",C
Input "D=",D
Input "E=",E
Input "F=",F

{A,B,C}→L₁
{D,E,F}→L₂

SortA(L₁)
SortA(L₂)

L₁(1)→A
L₁(2)→B
L₁(3)→C

L₂(1)→D
L₂(2)→E
L₂(3)→F

If abs(A*E-B*D)<.001 and abs(B*F-C*E)<.001
Then
Disp "SIM"
Else
Disp "NS"
End
Stop
