Menu("CHOOSE","IDENTIFY",1,"SIMILARITY",2)

Lbl 1
ClrHome
Prompt A,B,C
{A,B,C}→L₁
SortA(L₁)
L₁(1)→A
L₁(2)→B
L₁(3)→C
(A²+B²)→D

If A+B≤C
Then
Disp "NOT A TRIANGLE"
Else
If D=C²
Then
Disp "RIGHT"
Else
If D<C²
Then
Disp "OBTUSE"
Else
Disp "ACUTE"
End
End
End

If A=C
Then
Disp "EQUILATERAL"
Else
If A=B or B=C
Then
Disp "ISOSCELES"
Else
Disp "SCALENE"
End
End
Stop

Lbl 2
ClrHome
Disp "ENTER 0 IF","UNKNOWN"
Disp "TRI 1 SIDES"
Input "A:",A
Input "B:",B
Input "C:",C
Disp "TRI 2 SIDES"
Input "D:",D
Input "E:",E
Input "F:",F

0→S

If A>0 and B>0 and C>0 and D>0 and E>0 and F>0
Then
{A,B,C}→L₁
{D,E,F}→L₂
SortA(L₁)
SortA(L₂)
L₁(1)→A
L₁(2)→B
L₁(3)→C
L₂(1)→D
L₂(2)→E
L₂(3)→F
If abs(A*E-B*D)<.001 and abs(B*F-C*E)<.001
Then
1→S
Disp "SSS MATCH"
End
End

If S=1
Then
Disp "SIMILAR"
Else
Disp "NOT SIMILAR"
End
Stop
